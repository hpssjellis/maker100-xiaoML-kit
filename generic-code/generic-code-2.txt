int sensorLogs[] = {12, 45, 82, 14, 60};
int logCount = 5;
int alertThreshold = 50;
bool systemActive = true;


void printReport(int count) {
  Serial.print("Total items above threshold: ");
  Serial.println(count);
  if (count >= 3) {
    Serial.println("RESULT: High Activity Detected.");
  } else {
    Serial.println("RESULT: Normal Activity.");
  }
}


void processLogs() {
  int highReadingCount = 0;
  for (int i = 0; i < logCount; i++) {
    Serial.print("Checking Index ");
    Serial.print(i);
    Serial.print(": ");
    Serial.println(sensorLogs[i]);
    if (sensorLogs[i] > alertThreshold) {
      highReadingCount++;
    }
  }
  printReport(highReadingCount);
}


void setup() {
  Serial.begin(9600);
  delay(1000); 
  Serial.println("--- Booting Data Module ---");
  for (int i = 0; i < logCount; i++) {
    sensorLogs[i] = sensorLogs[i] + 5; 
  }
}


void loop() {
  if (systemActive) {
    processLogs();
    systemActive = false; 
    Serial.println("Analysis Complete.");
  }
}
