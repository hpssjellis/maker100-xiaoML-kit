#include "esp_camera.h"
#include "FS.h"
#include "SD.h"
#include "SPI.h"
#include <U8g2lib.h>
#include <Wire.h>

U8G2_SSD1306_72X40_ER_1_HW_I2C u8g2(U8G2_R2, U8X8_PIN_NONE);

// Timing for single-button logic
unsigned long myTouchStart = 0;
const int myLongPressTime = 200; // 200 ms is a "Select"
const int myShortInterval = 50;  // 50 ms is a next menu item"
bool myIsTouching = false;

int myMenuIndex = 1;
bool myIsSelected = false;
int myLastPicked = 0;
const int myThreshold = 1500; // Based on your A0 peak of 2000

void setup() {
  Serial.begin(115200);
  pinMode(A0, INPUT_PULLDOWN); // Use A0 since it's the stable one
  pinMode(LED_BUILTIN, OUTPUT);
  digitalWrite(LED_BUILTIN, HIGH); 
  u8g2.begin();
}

void loop() {
  int myA0Val = analogRead(A0);
  unsigned long myCurrentMillis = millis();

  if (myA0Val > myThreshold) {
    if (!myIsTouching) {
      myTouchStart = myCurrentMillis;
      myIsTouching = true;
      digitalWrite(LED_BUILTIN, LOW); // Feedback
    }
  } else {
    if (myIsTouching) {
      unsigned long myDuration = myCurrentMillis - myTouchStart;
      
      if (myDuration > myLongPressTime) {
        // LONG PRESS: SELECT or BACK
        if (myIsSelected) {
          myIsSelected = false;
        } else {
          myIsSelected = true;
          myLastPicked = myMenuIndex;
        }
      } else {
        // SHORT PRESS: CYCLE
        if (!myIsSelected) {
          myMenuIndex++;
          if (myMenuIndex > 3) myMenuIndex = 1;
          delay(myShortInterval);
        }
      }
      myIsTouching = false;
      digitalWrite(LED_BUILTIN, HIGH);
    }
  }

  // Draw Display
  u8g2.firstPage();
  do {
    u8g2.setFont(u8g2_font_6x10_tf);
    if (!myIsSelected) {
      u8g2.drawStr(0, 10, "-TAP:Next HOLD:Ok-");
      u8g2.drawStr(myMenuIndex == 1 ? 0 : 10, 22, (myMenuIndex == 1 ? "> Item 1" : "Item 1"));
      u8g2.drawStr(myMenuIndex == 2 ? 0 : 10, 31, (myMenuIndex == 2 ? "> Item 2" : "Item 2"));
      u8g2.drawStr(myMenuIndex == 3 ? 0 : 10, 40, (myMenuIndex == 3 ? "> Item 3" : "Item 3"));
    } else {
      u8g2.drawStr(0, 15, "You picked:");
      u8g2.setCursor(25, 38);
      u8g2.print(String(myLastPicked));
    }
  } while (u8g2.nextPage());
}
