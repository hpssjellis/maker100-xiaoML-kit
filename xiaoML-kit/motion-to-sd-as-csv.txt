#include <LSM6DS3.h>
#include <Wire.h>
#include "FS.h"
#include "SD.h"
#include "SPI.h"

LSM6DS3 myIMU(I2C_MODE, 0x6A);

const int TOUCH_PIN = A0; 
int touchThreshold = 0;
int sampleCount = 0;

void setup() {
  Serial.begin(115200);
  
  if (myIMU.begin() != 0) while (1);
  if (!SD.begin(21)) {
    Serial.println("SD card failed");
    while (1);
  }

  // Calibrate touch sensor (take average of 10 readings)
  long sum = 0;
  for(int i=0; i<10; i++) sum += touchRead(TOUCH_PIN);
  touchThreshold = (sum / 10) - 15; // Set threshold slightly below baseline

  Serial.println("Ready. Touch A0 to record 1 second of motion.");
}

void recordBurst() {
  char filename[32];
  sprintf(filename, "/motion_%d.csv", sampleCount++);
  
  File file = SD.open(filename, FILE_WRITE);
  if (!file) return;

  file.println("timestamp,accX,accY,accZ");
  
  Serial.printf("Recording to %s... ", filename);
  
  unsigned long startTime = millis();
  // Record for exactly 1000ms
  while (millis() - startTime < 1000) {
    file.print(millis() - startTime);
    file.print(",");
    file.print(myIMU.readFloatAccelX(), 4); file.print(",");
    file.print(myIMU.readFloatAccelY(), 4); file.print(",");
    file.println(myIMU.readFloatAccelZ(), 4);
    
    delay(10); // 100Hz capture rate
  }

  file.close();
  Serial.println("Done.");
  delay(500); // Debounce delay
}

void loop() {
  int touchValue = touchRead(TOUCH_PIN);
  
  // If the value drops significantly, the pin is touched
  if (touchValue < touchThreshold && touchValue > 0) {
    recordBurst();
  }
}
