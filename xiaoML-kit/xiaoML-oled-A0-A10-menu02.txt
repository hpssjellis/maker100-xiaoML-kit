#include <U8g2lib.h>
#include <Wire.h>

// Initialize the OLED display for XIAO ML Kit
U8G2_SSD1306_72X40_ER_1_HW_I2C u8g2(U8G2_R2, U8X8_PIN_NONE); 

unsigned long myPreviousMillis = 0;
const long myInterval = 150; // Interval for updates (e.g., 150ms)

int myMenuIndex = 1;         
bool myIsSelected = false;   
int myLastPicked = 0;        

void setup() {
  Serial.begin(115200);
  if (!Serial) { delay(5000); } // Wait 5 seconds for serial
  
  pinMode(LED_BUILTIN, OUTPUT);
  u8g2.begin();
  
  Serial.println("Menu System Initialized - A0 to Select");
}

void loop() {
  unsigned long myCurrentMillis = millis();

  // Read Touch Buttons
  int myA0Value = analogRead(A0);
  int myA10Value = analogRead(A10);

  // Logic for cycling menu (A10)
  if (myA10Value > 900) {
    digitalWrite(LED_BUILTIN, LOW); // LED ON
    if (!myIsSelected) {
      myMenuIndex++;
      if (myMenuIndex > 3) myMenuIndex = 1;
      Serial.println("Navigating to Item: " + String(myMenuIndex));
      delay(200); // Simple debounce
    } else {
      myIsSelected = false; // Return to menu if already showing a selection
      delay(200);
    }
  } 
  // Logic for selecting item (A0)
  else if (myA0Value > 900) {
    digitalWrite(LED_BUILTIN, LOW); // LED ON
    myIsSelected = true;
    myLastPicked = myMenuIndex;
    Serial.println("Selected Item: " + String(myLastPicked));
    delay(200);
  } 
  else {
    digitalWrite(LED_BUILTIN, HIGH); // LED OFF
  }

  // Display Update Logic using the drawing sequence loop
  if (myCurrentMillis - myPreviousMillis >= myInterval) {
    myPreviousMillis = myCurrentMillis;

    u8g2.firstPage();
    do {
      u8g2.setFont(u8g2_font_6x10_tf); 
      
      if (!myIsSelected) {
        u8g2.drawStr(0, 10, "--- MENU ---");
        
        // Render menu list with selection indicator
        if (myMenuIndex == 1) u8g2.drawStr(0, 22, "> Item 1");
        else u8g2.drawStr(10, 22, "Item 1");
        
        if (myMenuIndex == 2) u8g2.drawStr(0, 31, "> Item 2");
        else u8g2.drawStr(10, 31, "Item 2");
        
        if (myMenuIndex == 3) u8g2.drawStr(0, 40, "> Item 3");
        else u8g2.drawStr(10, 40, "Item 3");
        
      } else {
        // Selection Screen
        u8g2.setFont(u8g2_font_ncenB08_tr);
        u8g2.drawStr(0, 15, "You picked:");
        u8g2.setFont(u8g2_font_ncenB14_tr);
        u8g2.setCursor(25, 38);
        u8g2.print(String(myLastPicked));
      }
    } while (u8g2.nextPage());
  }
}
