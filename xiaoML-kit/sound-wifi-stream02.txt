#include "WiFi.h"
#include "ESP_I2S.h"
#include "WebServer.h"

const char* mySsid = "ssid2222";
const char* myPassword = "pass2222";

I2SClass myI2S;
WebServer myServer(80);

#define SAMPLE_RATE 16000
#define BUFFER_SAMPLES 512   // number of samples per chunk

// HTML
const char* myPage =
"<html><body style='text-align:center;'>"
"<h2>Live Audio</h2>"
"<audio controls autoplay src='/stream'></audio>"
"</body></html>";


// --- WAV HEADER ---
void mySendHeader(WiFiClient* myClient) {
  uint8_t myHeader[44];

  uint32_t mySampleRate = SAMPLE_RATE;
  uint16_t myBits = 16;
  uint16_t myChannels = 1;

  uint32_t myByteRate = mySampleRate * myChannels * myBits / 8;
  uint16_t myBlockAlign = myChannels * myBits / 8;

  uint32_t myDataSize = 0xFFFFFFFF;
  uint32_t myFileSize = myDataSize + 36;

  memcpy(myHeader, "RIFF", 4);
  memcpy(myHeader + 4, &myFileSize, 4);
  memcpy(myHeader + 8, "WAVE", 4);
  memcpy(myHeader + 12, "fmt ", 4);

  uint32_t myFmtSize = 16;
  memcpy(myHeader + 16, &myFmtSize, 4);

  uint16_t myFormat = 1;
  memcpy(myHeader + 20, &myFormat, 2);
  memcpy(myHeader + 22, &myChannels, 2);
  memcpy(myHeader + 24, &mySampleRate, 4);
  memcpy(myHeader + 28, &myByteRate, 4);
  memcpy(myHeader + 32, &myBlockAlign, 2);
  memcpy(myHeader + 34, &myBits, 2);

  memcpy(myHeader + 36, "data", 4);
  memcpy(myHeader + 40, &myDataSize, 4);

  myClient->write(myHeader, 44);
}


// --- ROOT ---
void myHandleRoot() {
  myServer.send(200, "text/html", myPage);
}


// --- STREAM ---
void myHandleStream() {

  WiFiClient myClient = myServer.client();

  myServer.setContentLength(CONTENT_LENGTH_UNKNOWN);
  myServer.send(200, "audio/wav", "");

  mySendHeader(&myClient);

  int16_t mySampleBuffer[BUFFER_SAMPLES];

  Serial.println("Streaming...");

  while (myClient.connected()) {

    // Fill buffer manually (THIS is the key fix)
    for (int i = 0; i < BUFFER_SAMPLES; i++) {

      int mySample = myI2S.read();

      // filter invalid samples (from wiki)
      if (mySample == -1 || mySample == 0 || mySample == 1) {
        mySample = 0;
      }

      mySampleBuffer[i] = mySample;
    }

    // send buffer as bytes
    myClient.write((uint8_t*)mySampleBuffer, BUFFER_SAMPLES * 2);

    delay(1);
  }

  Serial.println("Stopped");
}


// --- SETUP ---
void setup() {

  Serial.begin(115200);

  myI2S.setPinsPdmRx(42, 41);

  if (!myI2S.begin(I2S_MODE_PDM_RX, SAMPLE_RATE, I2S_DATA_BIT_WIDTH_16BIT, I2S_SLOT_MODE_MONO)) {
    Serial.println("I2S failed");
    while (1);
  }

  WiFi.begin(mySsid, myPassword);

  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }

  Serial.println("\nConnected");
  Serial.println(WiFi.localIP());

  myServer.on("/", myHandleRoot);
  myServer.on("/stream", myHandleStream);
  myServer.begin();
}


// --- LOOP ---
void loop() {
  myServer.handleClient();
}
