#include "esp_camera.h"
#include "FS.h"
#include "SD.h"
#include "SPI.h"

// Pin definitions for XIAO ESP32S3 [cite: 6, 18, 19, 20]
#define PWDN_GPIO_NUM     -1
#define RESET_GPIO_NUM    -1
#define XCLK_GPIO_NUM     10
#define SIOD_GPIO_NUM     40
#define SIOC_GPIO_NUM     39
#define Y9_GPIO_NUM       48
#define Y8_GPIO_NUM       11
#define Y7_GPIO_NUM       12
#define Y6_GPIO_NUM       14
#define Y5_GPIO_NUM       16
#define Y4_GPIO_NUM       18
#define Y3_GPIO_NUM       17
#define Y2_GPIO_NUM       15
#define VSYNC_GPIO_NUM    38
#define HREF_GPIO_NUM     47
#define PCLK_GPIO_NUM     13

// Custom configuration [cite: 7, 8, 35]
const int myTriggerPin = A0; 
int myImageCounter = 0;
bool myCameraReady = false;
bool mySdReady = false;

/**
 * Saves the current camera frame buffer to the SD card as a JPEG [cite: 9, 12, 13, 14, 15]
 */
void mySavePhotoToSd(const char * myPath) {
  camera_fb_t * myFrameBuffer = esp_camera_fb_get();
  
  if (!myFrameBuffer) {
    Serial.println("Failed to get camera frame buffer");
    return;
  }

  File myFile = SD.open(myPath, FILE_WRITE);
  if (!myFile) {
    Serial.println("Failed to open file for writing");
  } else {
    if (myFile.write(myFrameBuffer->buf, myFrameBuffer->len) == myFrameBuffer->len) {
      Serial.printf("Saved picture: %s\n", myPath);
    } else {
      Serial.println("Write failed");
    }
    myFile.close();
  }
  
  esp_camera_fb_return(myFrameBuffer);
}

void setup() {
  Serial.begin(115200);
  pinMode(myTriggerPin, INPUT);
  
  // Initialize Camera [cite: 17, 21, 23, 24, 27]
  camera_config_t myConfig;
  myConfig.ledc_channel = LEDC_CHANNEL_0;
  myConfig.ledc_timer = LEDC_TIMER_0;
  myConfig.pin_d0 = Y2_GPIO_NUM;
  myConfig.pin_d1 = Y3_GPIO_NUM;
  myConfig.pin_d2 = Y4_GPIO_NUM;
  myConfig.pin_d3 = Y5_GPIO_NUM;
  myConfig.pin_d4 = Y6_GPIO_NUM;
  myConfig.pin_d5 = Y7_GPIO_NUM;
  myConfig.pin_d6 = Y8_GPIO_NUM;
  myConfig.pin_d7 = Y9_GPIO_NUM;
  myConfig.pin_xclk = XCLK_GPIO_NUM;
  myConfig.pin_pclk = PCLK_GPIO_NUM;
  myConfig.pin_vsync = VSYNC_GPIO_NUM;
  myConfig.pin_href = HREF_GPIO_NUM;
  myConfig.pin_sscb_sda = SIOD_GPIO_NUM;
  myConfig.pin_sscb_scl = SIOC_GPIO_NUM;
  myConfig.pin_pwdn = PWDN_GPIO_NUM;
  myConfig.pin_reset = RESET_GPIO_NUM;
  myConfig.xclk_freq_hz = 20000000;
  myConfig.frame_size = FRAMESIZE_UXGA;
  myConfig.pixel_format = PIXFORMAT_JPEG;
  myConfig.grab_mode = CAMERA_GRAB_LATEST;
  myConfig.fb_location = CAMERA_FB_IN_PSRAM;
  myConfig.jpeg_quality = 10;
  myConfig.fb_count = 2;

  esp_err_t myErr = esp_camera_init(&myConfig);
  if (myErr == ESP_OK) {
    myCameraReady = true;
  } else {
    Serial.printf("Camera init failed: 0x%x\n", myErr);
  }

  // Initialize SD Card (XIAO ESP32S3 Sense uses Pin 21 for CS) [cite: 29, 30, 33]
  if (SD.begin(21)) {
    mySdReady = true;
    Serial.println("SD Card initialized.");
  } else {
    Serial.println("SD Card Mount Failed");
  }
  
  myImageCounter = random(1000, 9999);
}

void loop() {
  // Check if A0 pin is High and hardware is ready [cite: 35, 36]
  if (digitalRead(myTriggerPin) == HIGH && myCameraReady && mySdReady) {
    char myFileName[32];
    sprintf(myFileName, "/img_%d.jpg", myImageCounter);
    
    mySavePhotoToSd(myFileName);
    myImageCounter++;
    
    // Debounce/Delay to prevent multiple captures for one press
    delay(1000); 
  }
}
