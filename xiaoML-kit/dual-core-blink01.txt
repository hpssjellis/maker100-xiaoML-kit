// ============================================================
//  Dual-Core Teaching Demo — Seeed Studio XIAO ESP32-S3
//  Core 0 (app_cpu / "other core") : blink + read + print every 5 s
//  Core 1 (pro_cpu / "main core")  : blink + read + print every 3 s
// ============================================================
//
//  Key concepts demonstrated
//  --------------------------
//  1. xTaskCreatePinnedToCore()  — pin a task to a specific core
//  2. vTaskDelay()               — scheduler-friendly delay in tasks
//  3. Three mutexes, one per shared resource:
//       serialMutex — UART output
//       ledMutex    — LED_BUILTIN pin
//       adcMutex    — ADC peripheral
//  4. Build the full output String BEFORE taking serialMutex,
//     then send it in ONE Serial.println() call — this is what
//     actually prevents garbled output across cores.
//  5. Per-core counters — each core owns its own variable,
//     no mutex needed because only one core ever writes it.
//
// ============================================================

// ---------- Shared mutexes — one per shared resource --------
SemaphoreHandle_t serialMutex;
SemaphoreHandle_t ledMutex;
SemaphoreHandle_t adcMutex;

// ---------- Per-core counters — each core owns its own ------
//  No mutex needed: Core 0 only writes core0Count,
//  Core 1 only writes core1Count.  They never touch each other's.
volatile uint32_t core0Count = 0;
volatile uint32_t core1Count = 0;

// ============================================================
//  Helper: blink LED_BUILTIN for 50 ms (thread-safe)
// ============================================================
void blinkLED() {
  if (xSemaphoreTake(ledMutex, portMAX_DELAY)) {
    digitalWrite(LED_BUILTIN, HIGH);
    vTaskDelay(50 / portTICK_PERIOD_MS);
    digitalWrite(LED_BUILTIN, LOW);
    xSemaphoreGive(ledMutex);
  }
}

// ============================================================
//  Helper: thread-safe analogRead
//  The ESP32-S3 ADC peripheral is shared silicon — calling it
//  from two cores at once corrupts both results.
// ============================================================
int safeAnalogRead(uint8_t pin) {
  int value = 0;
  if (xSemaphoreTake(adcMutex, portMAX_DELAY)) {
    value = analogRead(pin);
    xSemaphoreGive(adcMutex);
  }
  return value;
}

// ============================================================
//  Helper: print one complete line (thread-safe)
//
//  KEY INSIGHT: build the entire String FIRST (no mutex held),
//  then grab serialMutex and send it in a SINGLE println().
//  Multiple Serial.print() calls inside a mutex block still
//  interleave because the UART ISR can fire between calls.
//  One atomic println() eliminates that window entirely.
// ============================================================
void safePrintln(const String &msg) {
  if (xSemaphoreTake(serialMutex, portMAX_DELAY)) {
    Serial.println(msg);
    xSemaphoreGive(serialMutex);
  }
}

// ============================================================
//  Task running on Core 0  — every 5 seconds
// ============================================================
void otherCoreTask(void * parameter) {
  vTaskDelay(500 / portTICK_PERIOD_MS);   // let Serial settle

  while (true) {
    core0Count++;                          // Core 0 owns this counter

    blinkLED();
    int adcValue = safeAnalogRead(A0);

    // Build the full line before touching the mutex
    String msg  = "[ Core 0 | count: " + String(core0Count) + " ] ";
    msg += "Hello from the OTHER core!  ";
    msg += "analogRead(A0) = " + String(adcValue);
    msg += "  (~" + String(adcValue * 3.3f / 4095.0f, 2) + " V)";
    msg += "  (core " + String(xPortGetCoreID()) + ")";

    safePrintln(msg);

    vTaskDelay(5000 / portTICK_PERIOD_MS);
  }
}

// ============================================================
//  setup() — runs once on Core 1
// ============================================================
void setup() {
  Serial.begin(115200);
  while (!Serial) { ; }
  delay(1000);

  Serial.println();
  Serial.println("============================================");
  Serial.println("  Dual-Core Demo — XIAO ESP32-S3");
  Serial.println("============================================");
  Serial.println("  setup() running on core: " + String(xPortGetCoreID()));
  Serial.println("--------------------------------------------");

  serialMutex = xSemaphoreCreateMutex();
  ledMutex    = xSemaphoreCreateMutex();
  adcMutex    = xSemaphoreCreateMutex();

  pinMode(LED_BUILTIN, OUTPUT);
  digitalWrite(LED_BUILTIN, LOW);

  xTaskCreatePinnedToCore(
    otherCoreTask,   // task function
    "OtherCore",     // name (shown in task list)
    4096,            // stack — increased to fit String building
    NULL,            // parameter
    1,               // priority
    NULL,            // handle (not needed)
    0                // <- Core 0
  );
}

// ============================================================
//  loop() — runs repeatedly on Core 1
// ============================================================
void loop() {
  core1Count++;                            // Core 1 owns this counter

  blinkLED();
  int adcValue = safeAnalogRead(A0);

  // Build the full line before touching the mutex
  String msg  = "[ Core 1 | count: " + String(core1Count) + " ] ";
  msg += "Hello from the MAIN core!  ";
  msg += "analogRead(A0) = " + String(adcValue);
  msg += "  (~" + String(adcValue * 3.3f / 4095.0f, 2) + " V)";
  msg += "  (core " + String(xPortGetCoreID()) + ")";

  safePrintln(msg);

  vTaskDelay(3000 / portTICK_PERIOD_MS);  // vTaskDelay works in loop() too
}
