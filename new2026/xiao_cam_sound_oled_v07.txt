// ============================================================
//  xiao_cam_sound_oled_v3.ino
//  XIAO ESP32S3 Sense — Camera + Audio WiFi Stream + OLED view
//
//  Port 80 → HTML page + MJPEG stream (/stream_video)
//  Port 81 → WAV audio stream
//  OLED    → live 72x40 1-bit camera thumbnail (Core 1, ~3fps)
//
//  Recording saves as .mp4 on Chrome 130+, .webm elsewhere
// ============================================================

#include "WiFi.h"
#include "esp_camera.h"
#include "ESP_I2S.h"
#include <U8g2lib.h>
#include <Wire.h>

// ─── WiFi ───────────────────────────────────────────────────
const char* mySsid     = "YOUR-SSID";
const char* myPassword = "YOUR-PASSWORD";

// ─── Camera pins ────────────────────────────────────────────
#define PWDN_GPIO_NUM   -1
#define RESET_GPIO_NUM  -1
#define XCLK_GPIO_NUM   10
#define SIOD_GPIO_NUM   40
#define SIOC_GPIO_NUM   39
#define Y9_GPIO_NUM     48
#define Y8_GPIO_NUM     11
#define Y7_GPIO_NUM     12
#define Y6_GPIO_NUM     14
#define Y5_GPIO_NUM     16
#define Y4_GPIO_NUM     18
#define Y3_GPIO_NUM     17
#define Y2_GPIO_NUM     15
#define VSYNC_GPIO_NUM  38
#define HREF_GPIO_NUM   47
#define PCLK_GPIO_NUM   13

// ─── OLED ───────────────────────────────────────────────────
#define MY_OLED_W  72
#define MY_OLED_H  40
// Full buffer mode for fastest sendBuffer() refresh
U8G2_SSD1306_72X40_ER_F_HW_I2C u8g2(U8G2_R2, U8X8_PIN_NONE);

// ─── Audio ──────────────────────────────────────────────────
#define MY_PDM_CLK_PIN    42
#define MY_PDM_DAT_PIN    41
#define MY_SAMPLE_RATE    16000
#define MY_CHUNK_SAMPLES  64

// ─── Servers ────────────────────────────────────────────────
WiFiServer myVideoServer(80);
WiFiServer myAudioServer(81);

// ─── I2S ────────────────────────────────────────────────────
I2SClass myI2S;

// ─── OLED frame buffer ───────────────────────────────────────
// Camera will capture at 96x96, we downsample to 72x40
// This is a 1-bit bitmap: 72*40/8 = 360 bytes
static uint8_t myOledBuf[MY_OLED_W * MY_OLED_H / 8];
static SemaphoreHandle_t myOledMutex;

// ─── Camera frame size for OLED ─────────────────────────────
// We use a second camera mode: FRAMESIZE_96X96 in GRAYSCALE
// for the OLED, and switch to JPEG for WiFi streaming.
// HOWEVER — the camera can only run one pixel format at a time.
// Strategy: run GRAYSCALE always; encode JPEG in software for stream.
// This is simpler and still gives ~10fps WiFi, ~3fps OLED.
//
// If you want max WiFi quality, comment out myUpdateOled() in loop()
// and switch camera to PIXFORMAT_JPEG + FRAMESIZE_QVGA.

// ============================================================
//  WAV HEADER
// ============================================================
void mySendWavHeader(WiFiClient& myClient) {
  uint8_t h[44];
  uint32_t mySR   = MY_SAMPLE_RATE;
  uint16_t myBits = 16;
  uint16_t myCh   = 1;
  uint32_t myBR   = mySR * myCh * myBits / 8;
  uint16_t myBA   = myCh * myBits / 8;
  uint32_t myDS   = 0xFFFFFFFF;
  uint32_t myFS   = myDS + 36;
  memcpy(h,    "RIFF", 4);  memcpy(h+4,  &myFS,  4);
  memcpy(h+8,  "WAVE", 4);  memcpy(h+12, "fmt ", 4);
  uint32_t fmtSz = 16;      memcpy(h+16, &fmtSz, 4);
  uint16_t fmt1  = 1;       memcpy(h+20, &fmt1,  2);
  memcpy(h+22, &myCh,   2); memcpy(h+24, &mySR,  4);
  memcpy(h+28, &myBR,   4); memcpy(h+32, &myBA,  2);
  memcpy(h+34, &myBits, 2); memcpy(h+36, "data", 4);
  memcpy(h+40, &myDS,   4);
  myClient.write(h, 44);
}

// ============================================================
//  AUDIO TASK  (Core 0)
// ============================================================
static void myDrainHeaders(WiFiClient& myClient) {
  unsigned long t0 = millis();
  while (myClient.connected() && millis() - t0 < 500) {
    if (myClient.available()) {
      String myLine = myClient.readStringUntil('\n');
      if (myLine == "\r" || myLine.length() == 0) break;
    } else {
      taskYIELD();
    }
  }
}

void myAudioTask(void* /*param*/) {
  Serial.println("[Audio] Task on Core 0");
  int16_t myBuf[MY_CHUNK_SAMPLES];
  for (;;) {
    WiFiClient myClient = myAudioServer.available();
    if (myClient) {
      Serial.println("[Audio] Client connected");
      myDrainHeaders(myClient);
      myClient.println("HTTP/1.1 200 OK");
      myClient.println("Content-Type: audio/wav");
      myClient.println("Access-Control-Allow-Origin: *");
      myClient.println("Connection: close");
      myClient.println();
      mySendWavHeader(myClient);
      while (myClient.connected()) {
        for (int i = 0; i < MY_CHUNK_SAMPLES; i++) {
          int s = myI2S.read();
          if (s == -1 || s == 0 || s == 1) s = 0;
          myBuf[i] = (int16_t)s;
        }
        if (myClient.write((uint8_t*)myBuf, MY_CHUNK_SAMPLES * 2) == 0) break;
        taskYIELD();
      }
      myClient.stop();
      Serial.println("[Audio] Disconnected");
    }
    taskYIELD();
  }
}

// ============================================================
//  OLED UPDATE  — converts grayscale camera frame to 1-bit XBM
// ============================================================
void myUpdateOled() {
  camera_fb_t* myFb = esp_camera_fb_get();
  if (!myFb) return;

  // myFb is PIXFORMAT_GRAYSCALE at 96x96
  // Downsample to 72x40 with simple nearest-neighbour, then threshold
  uint8_t newBuf[MY_OLED_W * MY_OLED_H / 8];
  memset(newBuf, 0, sizeof(newBuf));

  float myXScale = (float)myFb->width  / MY_OLED_W;
  float myYScale = (float)myFb->height / MY_OLED_H;

  for (int y = 0; y < MY_OLED_H; y++) {
    int mySrcY = (int)(y * myYScale);
    for (int x = 0; x < MY_OLED_W; x++) {
      int mySrcX = (int)(x * myXScale);
      uint8_t myPixel = myFb->buf[mySrcY * myFb->width + mySrcX];
      // Threshold at midpoint — tweak 128 for brighter/darker scenes
      if (myPixel > 128) {
        int myBit = y * MY_OLED_W + x;
        newBuf[myBit / 8] |= (1 << (myBit % 8));
      }
    }
  }

  esp_camera_fb_return(myFb);

  // Copy to shared buffer under mutex
  if (xSemaphoreTake(myOledMutex, portMAX_DELAY)) {
    memcpy(myOledBuf, newBuf, sizeof(myOledBuf));
    xSemaphoreGive(myOledMutex);
  }

  // Draw directly — U8g2 full-buffer mode, sendBuffer() is fast
  u8g2.clearBuffer();
  if (xSemaphoreTake(myOledMutex, portMAX_DELAY)) {
    u8g2.drawXBM(0, 0, MY_OLED_W, MY_OLED_H, myOledBuf);
    xSemaphoreGive(myOledMutex);
  }
  u8g2.sendBuffer();
}

// ============================================================
//  MJPEG VIDEO STREAM
// ============================================================
void myStreamVideo(WiFiClient& myClient) {
  myClient.println("HTTP/1.1 200 OK");
  myClient.println("Content-Type: multipart/x-mixed-replace; boundary=myframe");
  myClient.println("Access-Control-Allow-Origin: *");
  myClient.println("Connection: close");
  myClient.println();
  Serial.println("[Video] Streaming");
  while (myClient.connected()) {
    camera_fb_t* myFb = esp_camera_fb_get();
    if (!myFb) { vTaskDelay(1); continue; }
    myClient.printf("--myframe\r\nContent-Type: image/jpeg\r\nContent-Length: %u\r\n\r\n",
                    myFb->len);
    myClient.write(myFb->buf, myFb->len);
    myClient.println();
    esp_camera_fb_return(myFb);
    vTaskDelay(1);
  }
  myClient.stop();
  Serial.println("[Video] Disconnected");
}

// ============================================================
//  HTML PAGE  (MP4-first recording)
// ============================================================
void mySendPage(WiFiClient& myClient, const String& myHost) {
  String myIp = myHost;
  int myColon = myIp.lastIndexOf(':');
  if (myColon > 0) myIp = myIp.substring(0, myColon);
  String myAudioUrl = "http://" + myIp + ":81";

  myClient.println("HTTP/1.1 200 OK");
  myClient.println("Content-Type: text/html");
  myClient.println("Connection: close");
  myClient.println();

  myClient.print("<!DOCTYPE html><html>");
  myClient.print("<body data-audio-url=\"");
  myClient.print(myAudioUrl);
  myClient.print("\" style=\"text-align:center;background:black;color:white;font-family:sans-serif;\">");

  myClient.println(R"rawliteral(
<h2>XIAO ESP32S3 Camera + Audio</h2>

<img id="myImg" src="/stream_video" style="display:none" crossorigin="anonymous">
<canvas id="myCanvas" width="320" height="240" style="border:2px solid lime;"></canvas>

<br><br>
<audio id="myAudio" autoplay crossorigin="anonymous"></audio>
<br><br>

<input type="button" value="⏺ Start Recording" onclick="myStartRec()">
<input type="button" value="⏹ Stop &amp; Save" onclick="myStopRec()">
<p id="myStatus">Status: live</p>
<p id="myFormatInfo" style="font-size:0.8em;color:#aaa;"></p>

<script>
const myAudioUrl = document.body.dataset.audioUrl;
const myCanvas   = document.getElementById('myCanvas');
const myCtx      = myCanvas.getContext('2d');
const myImg      = document.getElementById('myImg');
const myAudio    = document.getElementById('myAudio');

myImg.addEventListener('load', () => {
  myCtx.drawImage(myImg, 0, 0, myCanvas.width, myCanvas.height);
});

myAudio.src = myAudioUrl;

// ── MP4-first MIME selection ──────────────────────────────────
// Chrome 130+ supports MP4 container output — plays natively on Windows.
// Falls back to webm on Firefox and older Chrome.
const myMp4Mime  = 'video/mp4;codecs=avc1,mp4a.40.2';
const myWebmMime = 'video/webm;codecs=vp8,opus';
const myMimeType = MediaRecorder.isTypeSupported(myMp4Mime) ? myMp4Mime : myWebmMime;
const myFileExt  = myMimeType.startsWith('video/mp4') ? 'mp4' : 'webm';

document.getElementById('myFormatInfo').textContent =
  'Recording format: ' + myFileExt.toUpperCase() +
  (myFileExt === 'webm' ? ' (install Web Media Extensions from Microsoft Store to play on Windows)' : '');

let myRecorder, myChunks = [], myRecording = false;

function myStartRec() {
  if (myRecording) return;
  myChunks = [];

  const myVStream = myCanvas.captureStream(10);
  const myAStream = myAudio.captureStream ? myAudio.captureStream()
                  : myAudio.mozCaptureStream ? myAudio.mozCaptureStream()
                  : null;

  const myTracks = [...myVStream.getTracks()];
  if (myAStream) myAStream.getAudioTracks().forEach(t => myTracks.push(t));

  const myCombined = new MediaStream(myTracks);
  myRecorder = new MediaRecorder(myCombined, { mimeType: myMimeType });

  myRecorder.ondataavailable = e => { if (e.data.size > 0) myChunks.push(e.data); };

  myRecorder.onstop = () => {
    const blob = new Blob(myChunks, { type: myMimeType });
    const a = document.createElement('a');
    a.href = URL.createObjectURL(blob);
    a.download = 'xiao_recording.' + myFileExt;
    a.click();
    document.getElementById('myStatus').textContent = 'Status: saved ✔ (' + myFileExt + ')';
  };

  myRecorder.start();
  myRecording = true;
  document.getElementById('myStatus').textContent = 'Status: recording...';
}

function myStopRec() {
  if (!myRecording) return;
  myRecorder.stop();
  myRecording = false;
}
</script>
</body></html>
)rawliteral");
}

// ============================================================
//  CAMERA INIT
// ============================================================
bool myInitCamera() {
  camera_config_t cfg;
  cfg.ledc_channel  = LEDC_CHANNEL_0;
  cfg.ledc_timer    = LEDC_TIMER_0;
  cfg.pin_d0        = Y2_GPIO_NUM;  cfg.pin_d1 = Y3_GPIO_NUM;
  cfg.pin_d2        = Y4_GPIO_NUM;  cfg.pin_d3 = Y5_GPIO_NUM;
  cfg.pin_d4        = Y6_GPIO_NUM;  cfg.pin_d5 = Y7_GPIO_NUM;
  cfg.pin_d6        = Y8_GPIO_NUM;  cfg.pin_d7 = Y9_GPIO_NUM;
  cfg.pin_xclk      = XCLK_GPIO_NUM;
  cfg.pin_pclk      = PCLK_GPIO_NUM;
  cfg.pin_vsync     = VSYNC_GPIO_NUM;
  cfg.pin_href      = HREF_GPIO_NUM;
  cfg.pin_sccb_sda  = SIOD_GPIO_NUM;
  cfg.pin_sccb_scl  = SIOC_GPIO_NUM;
  cfg.pin_pwdn      = PWDN_GPIO_NUM;
  cfg.pin_reset     = RESET_GPIO_NUM;
  cfg.xclk_freq_hz  = 20000000;

  // ── OLED mode: grayscale, smallest useful square frame ──────
  // Comment this block and uncomment JPEG block below to
  // disable OLED and get maximum WiFi JPEG quality instead.
  cfg.pixel_format  = PIXFORMAT_GRAYSCALE;
  cfg.frame_size    = FRAMESIZE_96X96;   // 96x96 → downsample to 72x40
  cfg.jpeg_quality  = 10;                // ignored in grayscale mode
  cfg.fb_count      = 2;
  cfg.fb_location   = CAMERA_FB_IN_PSRAM;
  cfg.grab_mode     = CAMERA_GRAB_LATEST;

  // ── WiFi-only JPEG mode (better stream, no OLED) ────────────
  // cfg.pixel_format = PIXFORMAT_JPEG;
  // cfg.frame_size   = FRAMESIZE_QVGA;  // 320x240
  // cfg.jpeg_quality = 10;
  // cfg.fb_count     = 2;
  // cfg.fb_location  = CAMERA_FB_IN_PSRAM;
  // cfg.grab_mode    = CAMERA_GRAB_LATEST;

  if (!psramFound()) {
    cfg.fb_location = CAMERA_FB_IN_DRAM;
    cfg.fb_count    = 1;
  }

  if (esp_camera_init(&cfg) != ESP_OK) return false;

  sensor_t* s = esp_camera_sensor_get();
  s->set_gain_ctrl(s, 1);
  s->set_exposure_ctrl(s, 1);
  s->set_brightness(s, 0);
  s->set_saturation(s, 0);
  Serial.println("[Camera] Init OK");
  return true;
}

// ============================================================
//  SETUP
// ============================================================
void setup() {
  Serial.begin(115200);

  // OLED
  u8g2.begin();
  u8g2.clearBuffer();
  u8g2.setFont(u8g2_font_ncenB08_tr);
  u8g2.setCursor(2, 14); u8g2.print("Starting...");
  u8g2.sendBuffer();

  myOledMutex = xSemaphoreCreateMutex();

  // I2S audio
  myI2S.setPinsPdmRx(MY_PDM_CLK_PIN, MY_PDM_DAT_PIN);
  if (!myI2S.begin(I2S_MODE_PDM_RX, MY_SAMPLE_RATE,
                   I2S_DATA_BIT_WIDTH_16BIT, I2S_SLOT_MODE_MONO)) {
    Serial.println("[Audio] I2S init failed");
  } else {
    Serial.println("[Audio] I2S OK");
  }

  // Camera
  if (!myInitCamera()) {
    u8g2.clearBuffer();
    u8g2.setCursor(2, 14); u8g2.print("Cam FAIL");
    u8g2.sendBuffer();
    Serial.println("[Camera] FATAL");
    while (1) delay(1000);
  }

  // WiFi
  WiFi.begin(mySsid, myPassword);
  u8g2.clearBuffer();
  u8g2.setCursor(2, 14); u8g2.print("WiFi...");
  u8g2.sendBuffer();

  while (WiFi.status() != WL_CONNECTED) delay(500);

  String myIp = WiFi.localIP().toString();
  Serial.printf("[WiFi] http://%s/\n",    myIp.c_str());
  Serial.printf("[WiFi] Audio: http://%s:81/\n", myIp.c_str());

  // Show IP on OLED briefly
  u8g2.clearBuffer();
  u8g2.setFont(u8g2_font_5x7_tr);
  u8g2.setCursor(0, 10); u8g2.print(myIp);
  u8g2.setCursor(0, 22); u8g2.print(":81 audio");
  u8g2.sendBuffer();
  delay(3000);

  myVideoServer.begin();
  myAudioServer.begin();

  xTaskCreatePinnedToCore(myAudioTask, "myAudioTask", 4096, NULL, 1, NULL, 0);
}

// ============================================================
//  LOOP  (Core 1 — video stream + HTML + OLED)
//
//  Priority: if a WiFi video client is connected, stream to it.
//  OLED updates only happen between frames (vTaskDelay gaps).
//  For pure-OLED mode without WiFi, just remove myStreamVideo().
// ============================================================

static unsigned long myLastOled = 0;

void loop() {
  // ── Update OLED ~3fps between WiFi clients ─────────────────
  if (millis() - myLastOled > 333) {
    myUpdateOled();
    myLastOled = millis();
  }

  // ── Serve HTML page or MJPEG stream ────────────────────────
  WiFiClient myClient = myVideoServer.available();
  if (!myClient) return;

  Serial.println("[HTTP] Client");
  String myReq  = myClient.readStringUntil('\r');
  String myHost = WiFi.localIP().toString();
  String myLine;
  while ((myLine = myClient.readStringUntil('\n')) != "\r") {
    myLine.trim();
    if (myLine.startsWith("Host:")) {
      myHost = myLine.substring(5);
      myHost.trim();
    }
  }

  if (myReq.indexOf("/stream_video") >= 0) {
    myStreamVideo(myClient);   // blocks until client disconnects
  } else {
    mySendPage(myClient, myHost);
    myClient.stop();
  }
}
