// ---- Camera Init ----
void myInitCamera() {
    camera_config_t myConfig;
    myConfig.ledc_channel = LEDC_CHANNEL_0;
    myConfig.ledc_timer   = LEDC_TIMER_0;
    myConfig.pin_d0       = myY2;
    myConfig.pin_d1       = myY3;
    myConfig.pin_d2       = myY4;
    myConfig.pin_d3       = myY5;
    myConfig.pin_d4       = myY6;
    myConfig.pin_d5       = myY7;
    myConfig.pin_d6       = myY8;
    myConfig.pin_d7       = myY9;
    myConfig.pin_xclk     = myXclk;
    myConfig.pin_pclk     = myPclk;
    myConfig.pin_vsync    = myVsync;
    myConfig.pin_href     = myHref;
    myConfig.pin_sscb_sda = mySiod;
    myConfig.pin_sscb_scl = mySioc;
    myConfig.pin_pwdn     = myPwdn;
    myConfig.pin_reset    = myReset;
    myConfig.xclk_freq_hz = 20000000;
    myConfig.pixel_format = PIXFORMAT_RGB565;
    myConfig.frame_size   = FRAMESIZE_240X240;
    myConfig.fb_count     = 1;                    // was 2 — saves ~115KB PSRAM
    myConfig.fb_location  = CAMERA_FB_IN_PSRAM;
    myConfig.grab_mode    = CAMERA_GRAB_WHEN_EMPTY;

    if (esp_camera_init(&myConfig) != ESP_OK) {
        Serial.println("ERROR: Camera init failed");
        return;
    }

    sensor_t *mySensor = esp_camera_sensor_get();
    if (mySensor) {
        mySensor->set_hmirror(mySensor, 0);
        mySensor->set_vflip(mySensor, 0);
        mySensor->set_brightness(mySensor, 1);
        mySensor->set_contrast(mySensor, 1);
        mySensor->set_saturation(mySensor, 1);
        mySensor->set_whitebal(mySensor, 1);
        mySensor->set_awb_gain(mySensor, 1);
        mySensor->set_wb_mode(mySensor, 0);
    }
    Serial.println("Camera OK");
}

// ---- Touch ----
bool myCheckTouch() {
    if (digitalRead(myTouchInt) != LOW) return false;
    Wire.beginTransmission(myTouchI2C);
    if (Wire.endTransmission() != 0) return false;
    uint8_t myBuf[5];
    Wire.requestFrom(myTouchI2C, 5);
    for (int i = 0; i < 5; i++) myBuf[i] = Wire.read();
    if (myBuf[0] == 0x01) {
        myTouchX = myBuf[2];
        myTouchY = myScreenHeight - myBuf[4];
        return true;
    }
    return false;
}

// ---- Setup ----
void setup() {
    Serial.begin(115200);
    delay(500);   // let USB CDC settle on ESP32-S3 before any SPI/I2C

    Serial.printf("Free PSRAM at boot: %u bytes\n", ESP.getFreePsram());
    Serial.printf("Free heap  at boot: %u bytes\n", ESP.getFreeHeap());

    // 1. I2C first (touch controller shares bus, must be up before TFT on some revisions)
    pinMode(myTouchInt, INPUT_PULLUP);
    Wire.begin();
    Serial.println("Wire OK");

    // 2. Camera before sprite — claims PSRAM early and cleanly
    myInitCamera();
    Serial.printf("Free PSRAM after camera: %u bytes\n", ESP.getFreePsram());

    // 3. Allocate snapshot buffer
    snapshot_buf = (uint8_t*)ps_malloc(myScreenWidth * myScreenHeight * 3);
    if (!snapshot_buf) {
        Serial.println("ERROR: snapshot_buf PSRAM alloc failed");
    } else {
        Serial.printf("snapshot_buf OK (%u bytes)\n", myScreenWidth * myScreenHeight * 3);
    }
    Serial.printf("Free PSRAM after snapshot_buf: %u bytes\n", ESP.getFreePsram());

    // 4. TFT last — SPI init after PSRAM is settled
    myTft.init();
    myTft.setRotation(1);
    myTft.setSwapBytes(true);
    Serial.println("TFT OK");

    // 5. Sprite — with explicit null check
    myFace.setColorDepth(16);
    void *spritePtr = myFace.createSprite(myScreenWidth, myScreenHeight);
    if (!spritePtr) {
        Serial.println("ERROR: Sprite allocation failed — check PSRAM");
        myDisplayReady = false;
    } else {
        myDisplayReady = true;
        Serial.printf("Sprite OK (%u bytes)\n", myScreenWidth * myScreenHeight * 2);
    }

    Serial.printf("Free PSRAM after sprite:  %u bytes\n", ESP.getFreePsram());
    Serial.printf("Free heap  after sprite:  %u bytes\n", ESP.getFreeHeap());
    Serial.println("Setup complete");
}
